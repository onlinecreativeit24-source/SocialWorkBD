import { getAuth, createUserWithEmailAndPassword } from "firebase/auth"; import { getFirestore, doc, setDoc } from "firebase/firestore"; const auth = getAuth(); const db = getFirestore(); async function handleSignUp(email, password, fullName, accountType) { try { // 1. Create User Authentication const userCredential = await createUserWithEmailAndPassword(auth, email, password); const user = userCredential.user; // 2. Select Target Collection Based on User Role let collectionName = "workers"; if (accountType === "Client") { collectionName = "clients"; } // 3. Save User Profile Data in Firestore await setDoc(doc(db, collectionName, user.uid), { uid: user.uid, fullName: fullName, email: email, accountType: accountType, createdAt: new Date() }); // Save in general users collection for common authentication rules await setDoc(doc(db, "users", user.uid), { uid: user.uid, email: email, accountType: accountType }); // 4. Direct Redirect Based on Selected Role if (accountType === "Client") { window.location.href = "client-dashboard.html"; } else { window.location.href = "worker-dashboard.html"; } } catch (error) { console.error("Sign Up Error:", error.message); alert("Error signing up: " + error.message); } }
